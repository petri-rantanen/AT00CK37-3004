/**
 * Generates the truth table for the logical AND operation.
 * Each row represents: [inputA, inputB, result]
 * @returns {Array<Array<boolean>>} Truth table for AND:
 * [
 *   [false, false, false],
 *   [false, true, false],
 *   [true, false, false],
 *   [true, true, true]
 * ]
 */
export function andTable() {
  return [
    [false, false, false],
    [false, true, false],
    [true, false, false],
    [true, true, true]
  ];
}
